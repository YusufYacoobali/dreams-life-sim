function sourceHash() {
  const digest = createHash('sha256');
  const visit = directory => { for (const entry of fs.readdirSync(directory, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    const file = path.join(directory, entry.name);
    if (entry.isDirectory()) visit(file);
    else if (entry.name.endsWith('.ts')) digest.update(path.relative(root, file)).update('\0').update(fs.readFileSync(file)).update('\0');
  } };
  visit(path.join(root, 'src/game')); visit(path.join(root, 'src/persistence')); return digest.digest('hex');
}
