import argparse,datetime,hashlib,json,pathlib,subprocess,sys,time
p=argparse.ArgumentParser()
p.add_argument('--root',required=True);p.add_argument('--source',required=True);p.add_argument('--prefix',required=True);p.add_argument('command',nargs=argparse.REMAINDER)
a=p.parse_args(); command=a.command[1:] if a.command and a.command[0]=='--' else a.command
assert command and a.prefix.startswith('/tmp/')
root=pathlib.Path(a.root);source=pathlib.Path(a.source);raw=pathlib.Path(a.prefix+'.out');receipt=pathlib.Path(a.prefix+'-receipt.json')
assert not raw.exists() and not receipt.exists(), 'Gate outputs must be new'
sha=lambda b:hashlib.sha256(b).hexdigest()
snapshot=json.loads(source.read_text()); assert snapshot['root']==str(root)
def verify():
 for f,h in snapshot['source'].items(): assert sha((root/f).read_bytes())==h, 'Source changed: '+f
 return {'files':len(snapshot['source']),'snapshotSha256':sha(source.read_bytes()),'status':'exact'}
before=verify();started=datetime.datetime.now(datetime.timezone.utc).isoformat();clock=time.monotonic()
record={'format':'dreams-coordinator-gate-execution-v1','sourceCommit':snapshot['commit'],'sourceSnapshot':str(source),'sourceSnapshotSha256':before['snapshotSha256'],'root':str(root),'argv':command,'startedAt':started,'before':before,'complete':False}
receipt.write_text(json.dumps(record,indent=2)+'\n')
with raw.open('xb') as out:
 result=subprocess.run(command,cwd=root,stdout=out,stderr=subprocess.STDOUT)
record.update({'endedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'elapsedSeconds':round(time.monotonic()-clock,3),'exitCode':result.returncode,'raw':{'file':str(raw),'bytes':raw.stat().st_size,'sha256':sha(raw.read_bytes())}})
try:record['after']=verify();record['complete']=True
except Exception as error:record['sourceVerificationError']=str(error)
receipt.write_text(json.dumps(record,indent=2)+'\n');print(json.dumps(record),flush=True)
sys.exit(result.returncode if result.returncode else (0 if record['complete'] else 2))
