def pin(path):
    path = Path(path)
    before = path.stat()
    raw = path.read_bytes()
    after = path.stat()
    assert (before.st_size, before.st_mtime_ns) == (after.st_size, after.st_mtime_ns), f'Changing input: {path}'
    return {'file': str(path), 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}

def add(path, role):
    path = str(path)
    if not Path(path).is_file():
        missing.append({'file': path, 'role': role})
        return None
    measured = pin(path)
    if path in d['artifacts']:
        assert all(d['artifacts'][path][field] == measured[field] for field in ['file', 'bytes', 'sha256']), f'Conflicting pin: {path}'
    else:
        d['artifacts'][path] = measured
    roles[path] = role
    return measured

def collect_pins(value, origin):
    if isinstance(value, dict):
        if {'file', 'bytes', 'sha256'} <= value.keys():
            path = Path(value['file'])
            if path.suffix in ['.db', '.db-wal', '.hbc', '.map'] or path.name.endswith('.db-wal'):
                assert path.is_file() and path.stat().st_size == value['bytes'], f'Missing external captured artifact: {path}'
                record = {key: value[key] for key in ['file', 'bytes', 'sha256']}
                record.update({'storage': 'external-only', 'pinBasis': origin + '; supplied immutable capture/proof pin, size checked only by this index'})
                if str(path) in external:
                    assert all(external[str(path)][key] == record[key] for key in ['file', 'bytes', 'sha256'])
                else:
                    external[str(path)] = record
            else:
                assert pin(path) == {key: value[key] for key in ['file', 'bytes', 'sha256']}, f'Captured pin mismatch: {path}'
                add(path, 'completed custom native compact evidence or exact source/tool proof')
        for child in value.values():
            collect_pins(child, origin)
    elif isinstance(value, list):
        for child in value:
            collect_pins(child, origin)
