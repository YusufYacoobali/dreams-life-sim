from pathlib import Path
from collections import Counter
from datetime import datetime, timezone
import json, hashlib, re

BASE = Path('/workspace/.cache/dreams-current-c39-policies/entrepreneur-v3')
OUT = Path(__file__).parent
read = lambda p: json.loads(Path(p).read_text())
sha = lambda b: hashlib.sha256(b).hexdigest()
pin = lambda p: {'file': str(p), 'bytes': Path(p).stat().st_size, 'sha256': sha(Path(p).read_bytes())}
canonical = lambda j: json.dumps(j, sort_keys=True, ensure_ascii=False, separators=(',', ':')).encode()
def write(name, obj):
    (OUT / name).write_text(json.dumps(obj, indent=2, ensure_ascii=False) + '\n')

receipt_path = Path('/tmp/dreams-production-finish-accepted-entrepreneur-v3-receipt.json')
receipt = read(receipt_path)
snapshot_path = Path(receipt['sourceSnapshot'])
snapshot = read(snapshot_path)
manifest = read(BASE / 'manifest.json')
aggregate = read(BASE / 'result.json')
registration_path = Path(manifest['source']['registration']['file'])

assert receipt['sourceCommit'] == snapshot['commit'] == '33b6a5fcb0aae73dfc235a15394b34d8afda00cc'
assert pin(snapshot_path)['sha256'] == receipt['sourceSnapshotSha256'] == '2e08fd925c673437ff7de7c76c0fd79d02832617d0014efc1c3e471782fe9b56'

    assert receipt[phase] == {'files': 688, 'snapshotSha256': receipt['sourceSnapshotSha256'], 'status': 'exact'}

expected_sources = {'game': snapshot['gameSourceHash'], 'gameAndPersistence': snapshot['combinedGamePersistenceSourceHash'], 'store': snapshot['source']['src/state/gameStore.ts'], 'tool': snapshot['source']['tools/current-policy/run-entrepreneur-v3.cjs']}

country_rows, gap_rows = [], []

for i, assignment in enumerate(assignments):
    c, seed = assignment['country'], assignment['seed']
    directory = BASE / c
    assert {p.name for p in directory.iterdir()} == {'birth.db', 'commands.ndjson', 'final.checkpoint.json', 'result.json'}
    r = read(directory / 'result.json')
    commands = [json.loads(line) for line in (directory / 'commands.ndjson').read_text().splitlines()]
    assert r == aggregate['records'][i] and r['country'] == c and r['seed'] == seed and r['versions'] == versions
    assert len(commands) == r['commands'] and sha((directory / 'commands.ndjson').read_bytes()) == r['traceSha256']

    assert commands[1]['command'] == {'type': 'acknowledgeBirth'}
    for j, row in enumerate(commands[2:], start=2):

        assert row['date']['year'] >= 2000

    assert r['birthProof']['closedConnectionReopenedExact'] is True and r['birthProof']['revision'] == 1
    assert r['birthProof']['databaseFile'] == str(directory / 'birth.db')
    assert all(re.fullmatch('[0-9a-f]{64}', r['birthProof'][key]) for key in ['worldSha256', 'allEightTablesSha256'])
    assert r['traceFile'] == str(directory / 'commands.ndjson') and r['finalCheckpointFile'] == str(directory / 'final.checkpoint.json')
    final_ref = pin(directory / 'final.checkpoint.json')
    final = r['finalCheckpointProof']
    assert final['checkpointSha256'] == final_ref['sha256'] and final['checkpointBytes'] == final_ref['bytes'] and final['worldSha256'] == r['finalWorldSha256']
    frame = read(directory / 'final.checkpoint.json')
    assert frame['format'] == final['format'] == 'dreams-checkpoint-v4'
    assert frame['bytes'] == final['compactBytes'] == frame['main']['bytes'] + sum(part['bytes'] for part in frame['people']) + sum(part['bytes'] for part in frame['ledger'])
    assert frame['bytes'] + 32 <= 32_000_000 and final['canonicalBytes'] <= 80_000_000
    assert final['canonicalBytes'] <= r['maximumCanonicalBytes'] and final['checkpointBytes'] <= r['maximumCheckpointBytes']
    cold_labels.update(r['productionCheckpointChecks']); proof_rows += len(r['coldProof'])
    assert all(proof['revision'] <= commands[-1]['revision'] and proof['canonicalBytes'] <= r['maximumCanonicalBytes'] and proof['checkpointBytes'] <= r['maximumCheckpointBytes'] for proof in r['coldProof'])
    for label, count in r['productionCheckpointChecks'].items():
        assert type(count) is int and count > 0
        assert sum(proof['label'] == label for proof in r['coldProof']) == (count if label in ['annual', 'death'] else 1)
    assert r['coldProof'][-1]['label'] == 'death' and {k: v for k, v in r['coldProof'][-1].items() if k not in ['label', 'revision']} == final

    assert [row['phase'] for row in r['timeline']] == PHASES

    assert r['coverage'] == {'allPhasesReached': True, 'orderedScenarioCComplete': True, 'reached': PHASES, 'unreached': []} and r['unmet'] == []
    for row in r['timeline']:

        if row['phase'] != 'birth': assert command['date'] == row['date'] and command['age'] == row['age']
    first = {row['phase']: row for row in r['timeline']}
    assert first['social']['baselineFollowers'] == r['social']['baselineFollowers'] == 0 and first['fame50']['fame'] >= 50
    assert commands[first['social']['commandIndex']]['command']['type'] == 'create-social'

    assert [rep['date'] for rep in r['reputation']] == [row['date'] for row in reputation_commands]
    funding = [row for row in selected_commands if row['command']['type'] == 'fund-business']

    country_rows.append({**assignment, 'ageAtDeath': r['ageAtDeath'], 'causeOfDeath': r['causeOfDeath'], 'timeline': r['timeline'], 'coverage': r['coverage'], 'commandsIncludingTwoBirthStoreSteps': r['commands'], 'pureEngineCommands': r['commands'] - 2, 'manualPosts': r['manualPosts'], 'viralPosts': r['social']['viralPosts'], 'peakFollowers': r['social']['peakFollowers'], 'socialIncomeMinor': r['social']['income'], 'currency': unit, 'reputationActions': len(r['reputation']), 'companyFundingMinor': r['companyFunding'], 'companyProfitMinor': company['cumulativeProfit'], 'companyDividendsMinor': company['dividends'], 'finalCompanyStatus': company['status'], 'lastAnnualCompanyStatus': r['annual'][-1]['company']['status'] if r['annual'][-1]['company'] else None, 'brandStatuses': dict(Counter(d['status'] for d in r['sponsorships'])), 'brandEarnedMinor': str(earned), 'children': r['childIds'], 'investmentFinalQuantity': r['holdings'][0]['quantity'], 'estateReceiptPairs': receipts['actualReceiptPairs'], 'producerColdComparisons': sum(r['productionCheckpointChecks'].values()), 'materializedColdProofRows': len(r['coldProof']), 'additionalFinalCheckpointRoundTrip': 1, 'maximumCanonicalBytes': r['maximumCanonicalBytes'], 'maximumCheckpointBytes': r['maximumCheckpointBytes'], 'traceSha256': r['traceSha256'], 'finalWorldSha256': r['finalWorldSha256'], 'raw': {name: pin(directory / name) for name in sorted(p.name for p in directory.iterdir())}})
