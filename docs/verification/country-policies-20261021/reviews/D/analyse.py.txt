read = lambda p: json.loads(Path(p).read_text())
sha = lambda b: hashlib.sha256(b).hexdigest()
canonical = lambda j: json.dumps(j, sort_keys=True, ensure_ascii=False, separators=(',', ':')).encode()
pin = lambda p: {'file': str(p), 'bytes': Path(p).stat().st_size, 'sha256': sha(Path(p).read_bytes())}
ndjson = lambda p: [json.loads(line) for line in Path(p).read_text().splitlines()]
def write(name, value):
    (ROOT / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')
def write_rows(name, rows):
    (ROOT / name).write_text(''.join(json.dumps(row, ensure_ascii=False, sort_keys=True) + '\n' for row in rows))
