function verifySource() {
  const digest = crypto.createHash('sha256');

    const bytes = fs.readFileSync(path.join(source, file));

    assert.equal(oid, sourceTree.get(file), 'Source differs from frozen Git blob: ' + file);
    digest.update(file + '\0'); digest.update(bytes); digest.update('\0');
  }
  assert.equal(digest.digest('hex'), plan.source.runtimeSourceSha256, 'Source differs from frozen projection');
}

function readCapture(file) {
  const db = new DatabaseSync(file, { readOnly: true });
  try {
    const actualTables = db.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name").all().map(row => row.name);
    assert.deepEqual(actualTables, tables.slice().sort(), file + ': exact eight tables');
    const rows = Object.fromEntries(tables.map(table => [table, db.prepare('SELECT * FROM ' + table).all()]));
    return { file, reader: db.prepare('PRAGMA user_version').get().user_version, rows, hashes: tableHash(rows) };
  } finally { db.close(); }
}
